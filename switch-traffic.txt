stage('Switch Traffic') {
    steps {
        script {
            echo "========== SWITCH TRAFFIC =========="

            sh '''
                docker exec blue-green-nginx sh -c '
                    sed "s|proxy_pass http://blue_backend;|proxy_pass http://green_backend;|g" \
                    /etc/nginx/conf.d/default.conf > /tmp/default.conf

                    mv -f /tmp/default.conf /etc/nginx/conf.d/default.conf

                    nginx -t

                    nginx -s reload

                    grep -q "proxy_pass http://green_backend;" \
                    /etc/nginx/conf.d/default.conf
                '
            '''

            echo "Traffic successfully switched to GREEN."

            sh '''
                docker exec blue-green-nginx \
                grep "proxy_pass" /etc/nginx/conf.d/default.conf
            '''
        }
    }
}
